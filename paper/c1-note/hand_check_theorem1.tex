  {\small Maxwell-Symmetry Defect of Stress-Nonlocal + Local-Heating Thermoelasticity in Helmholtz and Gibbs Representations}

\vspace{-4pt}
\noindent\rule{\textwidth}{0.6pt}

\paragraph{Claim (Theorem 1).}
For any $\ell > 0$ and $\gamma \in [0,1)$, the constitutive pairing that applies the nonlocal operator to the thermal stress,
\begin{equation}
  (1 - \ell^2\partial_x^2)\sigma = (1 - \gamma\ell^2\partial_x^2)(E\varepsilon - \beta\theta) \qquad \big(\text{at }\gamma = 0\text{: Class I, }(1-\ell^2\partial_x^2)\sigma = E\varepsilon - \beta\theta\big),\label{eq:1}
\end{equation}
while retaining the \emph{local} constant-strain entropy density
\begin{equation}
  \rho\eta^{\mathrm{loc}} = \beta\varepsilon + \frac{\rho c_E}{T_0}\theta,\label{eq:2}
\end{equation}
admits \textbf{(a)} no twice-differentiable Helmholtz free energy $\Psi[\varepsilon,\theta]$, \textbf{(b)} no stress-gradient Gibbs potential $G[\sigma,\theta]$ (where Gibbs reciprocity requires adding $+\frac{\beta\ell^2}{E}\partial_x^2\sigma$ to $\rho\eta^{\mathrm{loc}}$), and \textbf{(c)} yields negative internal dissipation $\Delta\mathcal{D}_{\mathrm{cycle}} < 0$ on closed harmonic cycles.

\smallskip\noindent\rule{\textwidth}{0.4pt}

\paragraph{Check 1: Helmholtz $(\varepsilon,\theta)$ Maxwell-Symmetry Defect.}
Take a single spatial Fourier mode $\varepsilon(x,t) = \hat{\varepsilon}(t)\cos(kx)$, $\theta(x,t) = \hat{\theta}(t)\cos(kx)$ ($k = n\pi/L > 0$) on $[0,L]$, where $-\partial_x^2[\cos(kx)] = k^2\cos(kx)$. Equations~\eqref{eq:1}--\eqref{eq:2} become algebraic in the modal amplitudes:
\begin{equation}
  \hat{\sigma}(\hat{\varepsilon},\hat{\theta}) = \frac{1 + \gamma k^2\ell^2}{1 + k^2\ell^2}\left(E\hat{\varepsilon} - \beta\hat{\theta}\right), \qquad
  \rho\hat{\eta}^{\mathrm{loc}}(\hat{\varepsilon},\hat{\theta}) = \beta\hat{\varepsilon} + \frac{\rho c_E}{T_0}\hat{\theta}.\label{eq:3}
\end{equation}
If a twice-differentiable modal Helmholtz free energy $\hat{\psi}(\hat{\varepsilon},\hat{\theta})$ existed with $\hat{\sigma} = \partial\hat{\psi}/\partial\hat{\varepsilon}$ and $\rho\hat{\eta}^{\mathrm{loc}} = -\partial\hat{\psi}/\partial\hat{\theta}$, Schwarz's theorem ($\partial^2\hat{\psi}/\partial\hat{\theta}\partial\hat{\varepsilon} = \partial^2\hat{\psi}/\partial\hat{\varepsilon}\partial\hat{\theta}$) would require $\frac{\partial\hat{\sigma}}{\partial\hat{\theta}} + \frac{\partial(\rho\hat{\eta}^{\mathrm{loc}})}{\partial\hat{\varepsilon}} = 0$. Differentiating \eqref{eq:3} by hand gives:
\begin{equation}
  \frac{\partial\hat{\sigma}}{\partial\hat{\theta}} = -\beta\,\frac{1 + \gamma k^2\ell^2}{1 + k^2\ell^2}, \quad
  \frac{\partial(\rho\hat{\eta}^{\mathrm{loc}})}{\partial\hat{\varepsilon}} = +\beta
  \implies
  \Delta_{\mathrm{Maxwell}}^{\mathrm{Helm}}(k) = \frac{\partial\hat{\sigma}}{\partial\hat{\theta}} + \frac{\partial(\rho\hat{\eta}^{\mathrm{loc}})}{\partial\hat{\varepsilon}} = \frac{\beta(1 - \gamma)k^2\ell^2}{1 + k^2\ell^2} \neq 0.\label{eq:4}
\end{equation}
\emph{(By contrast, Variant~V1 uses $\rho\hat{\eta}^{\mathrm{V1}} = \beta\frac{1+\gamma k^2\ell^2}{1+k^2\ell^2}\hat{\varepsilon} + \frac{\rho c_E}{T_0}\hat{\theta} \implies \Delta_{\mathrm{Maxwell}}^{\mathrm{V1}} = 0$, and Variant~V2 uses $\hat{\sigma}^{\mathrm{V2}} = E\frac{1+\gamma k^2\ell^2}{1+k^2\ell^2}\hat{\varepsilon} - \beta\hat{\theta}$ with $\rho\hat{\eta}^{\mathrm{V2}} = \rho\hat{\eta}^{\mathrm{loc}} \implies \Delta_{\mathrm{Maxwell}}^{\mathrm{V2}} = 0$.)}

\smallskip\noindent\rule{\textwidth}{0.4pt}

\paragraph{Check 2: Complementary Gibbs $(\sigma,\theta)$ Stress-Gradient Check at $\gamma = 0$.}
Suppose $(1 - \ell^2\partial_x^2)\sigma = E\varepsilon - \beta\theta$ is generated from the quadratic stress-gradient Gibbs functional $G[\sigma,\theta] = \int_0^L g(\sigma,\partial_x\sigma,\theta)\dd x$ with
\begin{equation}
  g(\sigma,\partial_x\sigma,\theta) = \frac{1}{2E}\sigma^2 + \frac{\ell^2}{2E}(\partial_x\sigma)^2 + \frac{\beta}{E}\sigma\theta + \frac{\rho c_\sigma}{2T_0}\theta^2, \qquad
  \frac{\rho c_\sigma}{T_0} := \frac{\rho c_E}{T_0} + \frac{\beta^2}{E}.\label{eq:5}
\end{equation}
\textbf{Step 1 (Strain law):} With $\partial_x\sigma = 0$ at $x = 0,L$, Euler--Lagrange variation with respect to $\sigma$ gives
\begin{equation}
  \varepsilon = \frac{\delta G}{\delta\sigma} = \frac{\partial g}{\partial\sigma} - \partial_x\!\left(\frac{\partial g}{\partial(\partial_x\sigma)}\right) = \frac{1}{E}(\sigma - \ell^2\partial_x^2\sigma) + \frac{\beta}{E}\theta \iff (1 - \ell^2\partial_x^2)\sigma = E\varepsilon - \beta\theta.\label{eq:6}
\end{equation}
\textbf{Step 2 (Mandatory Gibbs entropy law):} Substituting $\sigma = E\varepsilon - \beta\theta + \ell^2\partial_x^2\sigma$ from \eqref{eq:6} into $\rho\eta^{\mathrm{Gibbs}} = \partial g/\partial\theta$ gives
\begin{equation}
  \rho\eta^{\mathrm{Gibbs}} = \frac{\beta}{E}\sigma + \frac{\rho c_\sigma}{T_0}\theta = \frac{\beta}{E}\left(E\varepsilon - \beta\theta + \ell^2\partial_x^2\sigma\right) + \left(\frac{\rho c_E}{T_0} + \frac{\beta^2}{E}\right)\theta = \underbrace{\beta\varepsilon + \frac{\rho c_E}{T_0}\theta}_{\rho\eta^{\mathrm{loc}}} + \frac{\beta\ell^2}{E}\partial_x^2\sigma.\label{eq:7}
\end{equation}
The $-\frac{\beta^2}{E}\theta$ and $+\frac{\beta^2}{E}\theta$ terms cancel identically, leaving $\rho\eta^{\mathrm{Gibbs}} - \rho\eta^{\mathrm{loc}} = \frac{\beta\ell^2}{E}\partial_x^2\sigma \neq 0$.

\smallskip\noindent\rule{\textwidth}{0.4pt}

\paragraph{Check 3: Closed Harmonic Cycle Yielding Negative Internal Dissipation $\Delta\mathcal{D}_{\mathrm{cycle}} < 0$.}
Prescribe the $\mathcal{T} = 2\pi/\omega$-periodic history $\varepsilon(x,t) = \varepsilon_0\cos(kx)\cos(\omega t)$, $\theta(x,t) = \theta_0\cos(kx)\sin(\omega t)$ ($\varepsilon_0,\theta_0,\omega > 0$) on $[0,L]$. Over one period, any state function $\Psi$ has $\oint \dot{\Psi}\dd t = 0$. Using $\int_0^L \cos^2(kx)\dd x = \frac{L}{2}$, $\int_0^{\mathcal{T}}\cos(\omega t)\sin(\omega t)\dd t = 0$, and $\int_0^{\mathcal{T}}\sin^2(\omega t)\dd t = \int_0^{\mathcal{T}}\cos^2(\omega t)\dd t = \frac{\pi}{\omega}$:
\begin{align}
  \oint\int_0^L \sigma\dot{\varepsilon}\dd x\dd t &= \frac{L}{2}\int_0^{\mathcal{T}} \frac{1+\gamma k^2\ell^2}{1+k^2\ell^2}\left(E\varepsilon_0\cos\omega t - \beta\theta_0\sin\omega t\right)(-\omega\varepsilon_0\sin\omega t)\dd t = +\frac{\pi L}{2}\beta\varepsilon_0\theta_0\,\frac{1+\gamma k^2\ell^2}{1+k^2\ell^2},\label{eq:8}\\
  -\oint\int_0^L \rho\eta^{\mathrm{loc}}\dot{\theta}\dd x\dd t &= -\frac{L}{2}\int_0^{\mathcal{T}} \left(\beta\varepsilon_0\cos\omega t + \frac{\rho c_E}{T_0}\theta_0\sin\omega t\right)(\omega\theta_0\cos\omega t)\dd t = -\frac{\pi L}{2}\beta\varepsilon_0\theta_0.\label{eq:9}
\end{align}
Adding \eqref{eq:8}, \eqref{eq:9}, and the Fourier conduction dissipation $\oint\int_0^L \frac{\kappa}{T_0}(\partial_x\theta)^2\dd x\dd t = \frac{\pi L}{2\omega}\frac{\kappa k^2}{T_0}\theta_0^2$ yields
\begin{equation}
  \Delta\mathcal{D}_{\mathrm{cycle}} = \frac{\pi L\theta_0}{2\omega}\left[-\frac{\beta(1-\gamma)k^2\ell^2}{1 + k^2\ell^2}\,\omega\varepsilon_0 + \frac{\kappa k^2}{T_0}\,\theta_0\right] < 0 \quad \text{for } \varepsilon_0 > \frac{\kappa(1 + k^2\ell^2)}{\beta T_0(1-\gamma)\omega\ell^2}\,\theta_0. \quad \blacksquare\label{eq:10}
\end{equation}
